```text

├── Environments/
├── Datasets/          # 如 stack_the_base_llx_04251

```

## 3. pi05 训练示例

```bash
ROOT=~/H100_lerobot
HF="$ROOT/hf_offline"
IMG=lerobot:v052-cu128
mkdir -p "$ROOT/outputs"
OUT=pi05_stack_llx_$(date +%Y%m%d_%H%M)

docker run --rm -it --gpus all \
  -v "$ROOT/Datasets:/data/Datasets:ro" \
  -v "$ROOT/Models:/models:ro" \
  -v "$HF:/root/.cache/huggingface" \
  -v "$ROOT/outputs:/data/outputs" \
  -e HF_HUB_OFFLINE=1 \
  -e TRANSFORMERS_OFFLINE=1 \
  -e HF_DATASETS_OFFLINE=1 \
  "$IMG" \
  lerobot-train \
    --policy.type=pi05 \
    --policy.pretrained_path=/models/pi05_base \
    --dataset.repo_id=stack_the_base_llx_04251 \
    --dataset.root=/data/Datasets/stack_the_base_llx_04251 \
    --dataset.video_backend=pyav \
    --policy.device=cuda \
    --policy.dtype=bfloat16 \
    --policy.compile_model=false \
    --policy.gradient_checkpointing=true \
    --policy.train_expert_only=false \
    --policy.normalization_mapping='{"ACTION":"MEAN_STD","STATE":"MEAN_STD","VISUAL":"IDENTITY"}' \
    --batch_size=8 \
    --steps=30000 \
    --save_freq=5000 \
    --eval_freq=100000 \
    --num_workers=8 \
    --log_freq=100 \
    --output_dir=/data/outputs/$OUT \
    --job_name=$OUT \
    --wandb.enable=false \
    --policy.push_to_hub=false
```

检查点查看：

```bash
ls -lh "$ROOT/outputs/$OUT"
```

## 4. 常用参数

| 参数 | 说明 |
|------|------|
| `--dataset.video_backend=pyav` | **默认必加** |
| `--batch_size` | 冒烟用 1；正式可从 4/8 试起，OOM 则降低 |
| `--policy.gradient_checkpointing=true` | 省显存 |
| `--policy.dtype=bfloat16` | H100 推荐 |
| `--policy.train_expert_only` | `true` 更省显存；`false` 训全模型 |
| `--num_workers` | 可试 8～16 |
| `--gpus '"device=0"'` | 只用单卡时替换 `--gpus all` |
